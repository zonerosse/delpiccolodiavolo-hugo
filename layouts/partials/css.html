{{- /* Fogli di stile in un file a parte (Paolo, 18/10/2026, scelta C): il browser lo scarica una volta e lo tiene in memoria per tutte le pagine. Tre pacchetti: pagine chiare (main), pagine Notte (main+notte), Home (main+home+notte). Il nome contiene l'impronta del contenuto (fingerprint): cambiando il CSS cambia il nome, quindi la cache lunga è sicura. */ -}} {{- $main := resources.Get "css/main.css" -}} {{- if not $main }}{{ errorf "Il foglio di stile assets/css/main.css non esiste: la pagina uscirebbe senza CSS." }}{{ end -}} {{- $parti := slice $main -}} {{- $nome := "css/sito.css" -}} {{- $notte := or .Params.notte (and .IsPage (eq .Section "diario-allevamento")) -}} {{- if .IsHome }}{{ with resources.Get "css/home.css" }}{{ $parti = $parti | append . }}{{ end }}{{ $nome = "css/sito-home.css" }}{{ end -}} {{- if $notte }}{{ with resources.Get "css/notte.css" }}{{ $parti = $parti | append . }}{{ end }}{{ if not .IsHome }}{{ $nome = "css/sito-notte.css" }}{{ end }}{{ end -}} {{- $css := $parti | resources.Concat $nome | minify | fingerprint "sha256" -}}
